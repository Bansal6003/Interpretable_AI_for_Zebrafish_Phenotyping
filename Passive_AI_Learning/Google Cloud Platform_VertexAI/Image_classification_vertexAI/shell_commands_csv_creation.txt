# 1. Generate the CSV file directly by listing and appending labels
{
  gsutil ls gs://larva_orientation_classification/Lateral/* | grep -v '/$' | sed 's/$/,Lateral/'
  gsutil ls gs://larva_orientation_classification/Ventral/* | grep -v '/$' | sed 's/$/,Ventral/'
  gsutil ls gs://larva_orientation_classification/Ventro_Lateral/* | grep -v '/$' | sed 's/$/,Ventro_Lateral/'
} > class_orient_dataset.csv

# 2. Upload the finished CSV to your bucket
gsutil cp class_orient_dataset.csv gs://larva_orientation_classification/class_orient_dataset.csv
